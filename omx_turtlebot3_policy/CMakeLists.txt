cmake_minimum_required(VERSION 3.8)
project(omx_turtlebot3_policy)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(policy_probe_cpp src/policy_probe.cpp)
target_compile_features(policy_probe_cpp PRIVATE cxx_std_17)
target_compile_options(policy_probe_cpp PRIVATE -O2 -Wall -Wextra -Wpedantic)
target_link_libraries(policy_probe_cpp Eigen3::Eigen)
install(TARGETS policy_probe_cpp DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/policy_probe.py scripts/power_snapshot.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY models docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
