cmake_minimum_required(VERSION 3.16)
project(omx_turtlebot3_control_cpp)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(dynamixel_sdk REQUIRED)
find_package(urdf REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core dnn)
find_package(OpenSSL REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(omx_v6_control SHARED src/control.cpp src/opencr.cpp src/c_api.cpp)
target_compile_features(omx_v6_control PUBLIC cxx_std_17)
target_compile_options(omx_v6_control PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(omx_v6_control PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(omx_v6_control SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(omx_v6_control dynamixel_sdk urdf)
target_link_libraries(omx_v6_control ${OpenCV_LIBS} OpenSSL::Crypto yaml-cpp Threads::Threads)
add_executable(onnx_probe_cpp src/probe.cpp)
target_link_libraries(onnx_probe_cpp omx_v6_control)
install(TARGETS omx_v6_control EXPORT omx_v6_control LIBRARY DESTINATION lib)
install(TARGETS onnx_probe_cpp DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY docs DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_control test/test_control.cpp)
  target_link_libraries(test_control omx_v6_control)
  target_compile_definitions(test_control PRIVATE WORKSPACE_SRC="${CMAKE_CURRENT_SOURCE_DIR}/..")
endif()
ament_export_targets(omx_v6_control HAS_LIBRARY_TARGET)
ament_export_dependencies(dynamixel_sdk urdf OpenCV OpenSSL yaml-cpp Threads)
ament_package()
