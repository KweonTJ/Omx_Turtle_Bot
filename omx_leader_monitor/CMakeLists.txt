cmake_minimum_required(VERSION 3.16)
project(omx_leader_monitor)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(leader_monitor src/leader_monitor.cpp)
target_include_directories(leader_monitor PRIVATE include)
target_link_libraries(leader_monitor Qt5::Core Qt5::Gui Qt5::Widgets)
ament_target_dependencies(leader_monitor rclcpp sensor_msgs geometry_msgs std_msgs rviz_common rviz_rendering)
target_compile_options(leader_monitor PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS leader_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_leader_pick_place_monitor.sh DESTINATION share/${PROJECT_NAME}/scripts)
install(PROGRAMS scripts/demo_telemetry.py scripts/topic_bridge.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_monitor_model test/test_monitor_model.cpp)
  target_include_directories(test_monitor_model PRIVATE include)
  target_link_libraries(test_monitor_model Qt5::Core Qt5::Gui)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_monitor_bridge test/test_bridge.py)
endif()
ament_package()
